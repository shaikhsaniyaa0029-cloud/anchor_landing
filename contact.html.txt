<script>

  /* Mobile menu */

  const menuBtn = document.getElementById('menu-btn');
  const navLinks = document.getElementById('nav-links');

  menuBtn.addEventListener('click', () => {
    navLinks.classList.toggle('open');
  });

  navLinks.querySelectorAll('a').forEach(link => {

    link.addEventListener('click', () => {
      navLinks.classList.remove('open');
    });

  });


  /* Contact form validation */

  const form = document.getElementById('contact-form');
  const status = document.getElementById('form-status');

  const validators = {

    name: value =>
      value.trim().length >= 2,

    trade: value =>
      value.trim().length >= 2,

    email: value =>
      /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value.trim()),

    phone: value =>
      value.trim() === '' ||
      /^[0-9+()\-.\s]{7,}$/.test(value.trim()),

    message: value =>
      value.trim().length >= 10

  };


  function setFieldError(fieldElement, hasError) {

    fieldElement.classList.toggle(
      'has-error',
      hasError
    );

  }


  function validateField(name) {

    const fieldElement =
      form.querySelector(`[data-field="${name}"]`);

    const input =
      fieldElement.querySelector('input, textarea');

    const valid =
      validators[name](input.value);

    setFieldError(fieldElement, !valid);

    return valid;

  }


  Object.keys(validators).forEach(name => {

    const fieldElement =
      form.querySelector(`[data-field="${name}"]`);

    const input =
      fieldElement.querySelector('input, textarea');

    input.addEventListener('blur', () => {
      validateField(name);
    });

    input.addEventListener('input', () => {

      if (
        fieldElement.classList.contains('has-error')
      ) {
        validateField(name);
      }

    });

  });


  form.addEventListener('submit', event => {

    event.preventDefault();

    const results =
      Object.keys(validators).map(validateField);

    const allValid =
      results.every(Boolean);

    status.classList.remove(
      'show',
      'success',
      'fail'
    );


    if (!allValid) {

      status.textContent =
        'Please fix the highlighted fields before sending.';

      status.classList.add(
        'show',
        'fail'
      );

      const firstError =
        form.querySelector(
          '.has-error input, .has-error textarea'
        );

      if (firstError) {
        firstError.focus();
      }

      return;

    }


    status.textContent =
      "Thanks — we'll be in touch within one business day.";

    status.classList.add(
      'show',
      'success'
    );

    form.reset();

  });

</script>
